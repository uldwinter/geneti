function scoreProduct(p){
  if(!p)return{completeness:0,evidence:0,total:0,issues:[]};
  const required=[
    ['name','Название продукта'],['brand','Бренд'],['model','Модель / артикул'],['identifier','Уникальный идентификатор'],['manufacturer','Производитель'],['economicOperator','Экономический оператор'],['operatorContact','Контакт оператора'],['manufactureCountry','Страна производства'],['description','Описание'],['intendedUse','Назначение'],['repairInstructions','Инструкции по ремонту'],['recyclingInstructions','Инструкции по переработке']
  ];
  const optional=[['carbonFootprint','Углеродный след'],['warrantyMonths','Гарантия'],['sparePartsYears','Доступность запчастей'],['repairScore','Ремонтопригодность'],['expectedLifeYears','Срок службы'],['recyclablePct','Перерабатываемая доля'],['takeBackProgram','Программа возврата']];
  let points=0,max=required.length*5+optional.length*3+20;
  const issues=[];
  required.forEach(([key,label])=>{if(clean(p[key]))points+=5;else issues.push({severity:'critical',title:`Не заполнено: ${label}`,fix:`Добавьте поле «${label}» и укажите источник данных.`})});
  optional.forEach(([key,label])=>{if(clean(p[key])!=='' )points+=3;else issues.push({severity:'warning',title:`Нет показателя: ${label}`,fix:'Определите владельца данных и запросите подтверждающий документ.'})});
  if(p.materials?.length){points+=6;const sum=p.materials.reduce((a,m)=>a+(Number(m.share)||0),0);if(Math.abs(sum-100)<=1)points+=4;else issues.push({severity:'warning',title:`Сумма материалов составляет ${sum}%`,fix:'Уточните состав или объясните неизвестную долю.'});}else issues.push({severity:'critical',title:'Нет состава продукта',fix:'Добавьте материалы и массовые доли.'});
  if(p.suppliers?.length>=2)points+=5;else issues.push({severity:'warning',title:'Цепочка поставок неполна',fix:'Добавьте ключевых Tier 1/Tier 2 поставщиков и компоненты.'});
  if(p.documents?.length>=2)points+=5;else issues.push({severity:'warning',title:'Недостаточно доказательств',fix:'Добавьте декларации, LCA, сертификаты и инструкции.'});
  const verifiedDocs=(p.documents||[]).filter(d=>d.verified).length;
  const verifiedSuppliers=(p.suppliers||[]).filter(s=>s.evidence==='verified').length;
  const evidenceDenom=Math.max(1,(p.documents||[]).length+(p.suppliers||[]).length);
  const evidence=Math.round(((verifiedDocs+verifiedSuppliers)/evidenceDenom)*100);
  if(evidence<60)issues.push({severity:'warning',title:'Низкая доля проверенных доказательств',fix:'Проверьте документы и происхождение данных у поставщиков.'});
  if(!p.publicTitle)issues.push({severity:'info',title:'Публичная карточка не оформлена',fix:'Задайте публичный заголовок и сообщение бренда.'});
  const completeness=Math.min(100,Math.round(points/max*100));
  const total=Math.round(completeness*.72+evidence*.28);
  if(total>=85)issues.unshift({severity:'ok',title:'Паспорт имеет сильную основу',fix:'Проведите отраслевую юридическую проверку и подготовьте интеграцию с официальным реестром.'});
  return{completeness,evidence,total,issues};
}
